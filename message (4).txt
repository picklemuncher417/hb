---------------------------------------------------------
-- SLIDER LOGIC TRACKING
---------------------------------------------------------
local draggingSize = false
local draggingTrans = false

SliderButton.MouseButton1Down:Connect(function()
    draggingSize = true
end)

TransSliderButton.MouseButton1Down:Connect(function()
    draggingTrans = true
end)

game:GetService('UserInputService').InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        draggingSize = false
        draggingTrans = false
    end
end)

RS.RenderStepped:Connect(function()
    local mouse = Client:GetMouse()
    
    -- Handle Size Slider Dragging
    if draggingSize then
        local relativeX = math.clamp(
            mouse.X - SliderFrame.AbsolutePosition.X,
            0,
            SliderFrame.AbsoluteSize.X
        )
        local percentage = relativeX / SliderFrame.AbsoluteSize.X
        size = math.floor(5 + (percentage * 95))
        SliderButton.Position = UDim2.new(percentage, -20, 0, 0)
        SizeLabel.Text = 'Size: ' .. size
    end
    
    -- Handle Transparency Slider Dragging
    if draggingTrans then
        local relativeX = math.clamp(
            mouse.X - TransSliderFrame.AbsolutePosition.X,
            0,
            TransSliderFrame.AbsoluteSize.X
        )
        local percentage = relativeX / TransSliderFrame.AbsoluteSize.X
        hitBoxTransparency = percentage
        TransSliderButton.Position = UDim2.new(percentage, -20, 0, 0)
        TransLabel.Text = 'Transparency: ' .. math.round(hitBoxTransparency * 100) .. '%'
    end
end)

---------------------------------------------------------
-- VISUAL UPDATER LOOP
---------------------------------------------------------
RS.RenderStepped:Connect(function()
    for _, Player in pairs(Players:GetPlayers()) do
        if Player == Client then
            continue
        end

        if Player.Character and Player.Character:FindFirstChild('HumanoidRootPart') then
            local HRP = Player.Character.HumanoidRootPart
            local Humanoid = Player.Character:FindFirstChild('Humanoid')

            if Humanoid and Humanoid.Health > 0 then
                HRP.Size = Vector3.new(size, size, size)
                HRP.Transparency = hitBoxTransparency -- Directly tied to transparency slider
                HRP.CanCollide = false
                HRP.Color = Color3.fromRGB(255, 0, 0)

                -- Control edge lines visibility dynamically using the slider percentage
                local lineThickness = (1 - hitBoxTransparency) * 0.08
                local areLinesVisible = hitBoxTransparency < 1

                if not HRP:FindFirstChild('SelectionBox') then
                    local outline = Instance.new('SelectionBox')
                    outline.Name = 'SelectionBox'
                    outline.Parent = HRP
                    outline.Adornee = HRP
                    outline.LineThickness = lineThickness
                    outline.Visible = areLinesVisible
                    outline.Color3 = Color3.fromRGB(0, 0, 0)
                else
                    local outline = HRP:FindFirstChild('SelectionBox')
                    outline.LineThickness = lineThickness
                    outline.Visible = areLinesVisible
                end
            else
                HRP.Size = Vector3.new(2, 2, 1)
                HRP.Transparency = 1

                local outline = HRP:FindFirstChild('SelectionBox')
                if outline then
                    outline:Destroy()
                end
            end
        end
    end
end)

-- Toggle Menu Visibility (Using the N Key)
game:GetService('UserInputService').InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.N then
        ScreenGui.Enabled = not ScreenGui.Enabled
    end
end)

-- INPUT HANDLING AND INTERFACE CALCULATIONS
---------------------------------------------------------
local dragX, dragY, dragZ, dragTrans = false, false, false, false

ButtonX.MouseButton1Down:Connect(function() dragX = true end)
ButtonY.MouseButton1Down:Connect(function() dragY = true end)
ButtonZ.MouseButton1Down:Connect(function() dragZ = true end)
TransSliderButton.MouseButton1Down:Connect(function() dragTrans = true end)

game:GetService('UserInputService').InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragX, dragY, dragZ, dragTrans = false, false, false, false
    end
end)

RS.RenderStepped:Connect(function()
    local mouse = Client:GetMouse()
    local xTrackWidth = FrameX.AbsoluteSize.X
    
    if dragX then
        local pct = math.clamp(mouse.X - FrameX.AbsolutePosition.X, 0, xTrackWidth) / xTrackWidth
        sizeX = math.floor(1 + (pct * 99)) 
        ButtonX.Position = UDim2.new(pct, -10, 0.5, -10)
        LabelX.Text = 'Width (X): ' .. sizeX
    end
    if dragY then
        local pct = math.clamp(mouse.X - FrameY.AbsolutePosition.X, 0, FrameY.AbsoluteSize.X) / FrameY.AbsoluteSize.X
        sizeY = math.floor(1 + (pct * 99)) 
        ButtonY.Position = UDim2.new(pct, -10, 0.5, -10)
        LabelY.Text = 'Height (Y): ' .. sizeY
    end
    if dragZ then
        local pct = math.clamp(mouse.X - FrameZ.AbsolutePosition.X, 0, FrameZ.AbsoluteSize.X) / FrameZ.AbsoluteSize.X
        sizeZ = math.floor(1 + (pct * 99)) 
        ButtonZ.Position = UDim2.new(pct, -10, 0.5, -10)
        LabelZ.Text = 'Depth (Z): ' .. sizeZ
    end
    if dragTrans then
        local pct = math.clamp(mouse.X - TransSliderFrame.AbsolutePosition.X, 0, TransSliderFrame.AbsoluteSize.X) / TransSliderFrame.AbsoluteSize.X
        hitBoxTransparency = pct
        TransSliderButton.Position = UDim2.new(pct, -10, 0.5, -10)
        TransLabel.Text = 'Visibility Blockage: ' .. math.round((1 - hitBoxTransparency) * 100) .. '%'
    end
end)

ToggleButton.MouseButton1Click:Connect(function()
    hitboxEnabled = not hitboxEnabled
    if hitboxEnabled then
        ToggleButton.Text = 'Status: Active'
        ToggleButton.BackgroundColor3 = Color3.fromRGB(0, 170, 100)
        AccentLine.BackgroundColor3 = Color3.fromRGB(0, 170, 100)
    else
        ToggleButton.Text = 'Status: Disabled'
        ToggleButton.BackgroundColor3 = Color3.fromRGB(180, 50, 60) -- Muted red
        AccentLine.BackgroundColor3 = Color3.fromRGB(180, 50, 60)
    end
end)

---------------------------------------------------------
-- COMPREHENSIVE SIMULATION RUNTIME ENVIRONMENT
---------------------------------------------------------
RS.RenderStepped:Connect(function()
    for _, Player in pairs(Players:GetPlayers()) do
        if Player == Client then continue end

        if Player.Character and Player.Character:FindFirstChild('HumanoidRootPart') then
            local HRP = Player.Character.HumanoidRootPart
            local Humanoid = Player.Character:FindFirstChild('Humanoid')

            if Humanoid then
                local maxHealth = Humanoid.MaxHealth
                local currentHealth = Humanoid.Health

                if currentHealth <= 2 then
                    coolingDown[Player.UserId] = true
                elseif currentHealth >= maxHealth then
                    coolingDown[Player.UserId] = nil
                end

                if currentHealth > 2 and not coolingDown[Player.UserId] then
                    if hitboxEnabled then
                        HRP.Size = Vector3.new(sizeX, sizeY, sizeZ)
                    else
                        HRP.Size = Vector3.new(2, 2, 1)
                    end
                    
                    HRP.Transparency = hitBoxTransparency
                    HRP.CanCollide = false
                    HRP.Color = Color3.fromRGB(255, 0, 0)

                    local lineThickness = (1 - hitBoxTransparency) * 0.08
                    local areLinesVisible = hitBoxTransparency < 1

                    if not HRP:FindFirstChild('SelectionBox') then
                        local outline = Instance.new('SelectionBox')
                        outline.Name = 'SelectionBox'
                        outline.Parent = HRP
                        outline.Adornee = HRP
                        outline.LineThickness = lineThickness
                        outline.Visible = areLinesVisible
                        outline.Color3 = Color3.fromRGB(0, 0, 0)
                    else
                        local outline = HRP:FindFirstChild('SelectionBox')
                        outline.LineThickness = lineThickness
                        outline.Visible = areLinesVisible
                    end
                else
                    HRP.Size = Vector3.new(0, 0, 0)
                    HRP.Transparency = 1 

                    local outline = HRP:FindFirstChild('SelectionBox')
                    if outline then
                        outline:Destroy()
                    end
                end
            end
        end
    end
end)

game:GetService('UserInputService').InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.N then
        ScreenGui.Enabled = not ScreenGui.Enabled
    end
end)

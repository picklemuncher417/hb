local dragX, dragY, dragZ, dragTrans = false, false, false, false

ButtonX.MouseButton1Down:Connect(function() dragX = true end)

ButtonZ.MouseButton1Down:Connect(function() dragZ = true end)

game:GetService('UserInputService').InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragX, dragY, dragZ, dragTrans = false, false, false, false
    end
end)

-- Helper function to apply the transparency state instantly across players when dragging
local function applyInstantTransparency(value)
    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= Client and Player.Character then
            local HRP = Player.Character:FindFirstChild('HumanoidRootPart')
            local Humanoid = Player.Character:FindFirstChild('Humanoid')
            if HRP and Humanoid and Humanoid.Health > 2 and not recoveringPlayers[Player.UserId] then
                HRP.Transparency = value
                local outline = HRP:FindFirstChild('SelectionBox')
                if outline then
                    outline.LineThickness = (1 - value) * 0.08
                    outline.Visible = value < 1
                end
            end
        end
    end
end

RS.RenderStepped:Connect(function()
    local mouse = Client:GetMouse()
    
    if dragX then
        local pctX = math.clamp(mouse.X - FrameX.AbsolutePosition.X, 0, FrameX.AbsoluteSize.X) / FrameX.AbsoluteSize.X
        sizeX = math.floor(1 + (pctX * 99))
        ButtonX.Position = UDim2.new(pctX, -20, 0, 0)
        LabelX.Text = 'Size X: ' .. sizeX
    end
    if dragY then
        local pctY = math.clamp(mouse.X - FrameY.AbsolutePosition.X, 0, FrameY.AbsoluteSize.X) / FrameY.AbsoluteSize.X
        sizeY = math.floor(1 + (pctY * 99))
        ButtonY.Position = UDim2.new(pctY, -20, 0, 0)
        LabelY.Text = 'Size Y: ' .. sizeY
    end
    if dragZ then
        local pctZ = math.clamp(mouse.X - FrameZ.AbsolutePosition.X, 0, FrameZ.AbsoluteSize.X) / FrameZ.AbsoluteSize.X
        sizeZ = math.floor(1 + (pctZ * 99))
        ButtonZ.Position = UDim2.new(pctZ, -20, 0, 0)
        LabelZ.Text = 'Size Z: ' .. sizeZ
    end
    if dragTrans then
        local pctTrans = math.clamp(mouse.X - TransSliderFrame.AbsolutePosition.X, 0, TransSliderFrame.AbsoluteSize.X) / TransSliderFrame.AbsoluteSize.X
        hitBoxTransparency = pctTrans
        TransSliderButton.Position = UDim2.new(pctTrans, -20, 0, 0)
        TransLabel.Text = 'Transparency: ' .. math.round(hitBoxTransparency * 100) .. '%'
        applyInstantTransparency(hitBoxTransparency) -- Force transparency updates immediately
    end
end)

-- BACKEND EXECUTOR LOGIC WHEEL
---------------------------------------------------------
task.spawn(function()
    local baseSleep = 0.03
    while true do
        local currentSleep = baseSleep
        local activePlayers = Players:GetPlayers()
        local clientChar = Client.Character
        local clientHRP = clientChar and clientChar:FindFirstChild('HumanoidRootPart')
        
        if superOptimize and (#activePlayers <= 1 or not clientHRP) then
            currentSleep = 0.08
        end
        
        task.wait(currentSleep)
        
        for i = 1, #activePlayers do
            local Player = activePlayers[i]
            if Player == Client then continue end
            
            local character = Player.Character
            if character then
                local HRP = character:FindFirstChild('HumanoidRootPart')
                local Humanoid = character:FindFirstChild('Humanoid')
                
                if HRP and Humanoid then
                    if superOptimize and clientHRP then
                        local distance = (clientHRP.Position - HRP.Position).Magnitude
                        if distance > 400 then
                            continue
                        end
                    end
                    
                    local currentHealth = Humanoid.Health
                    
                    -- Cooldown trigger handling based on 2 HP and 40 HP rules
                    if currentHealth <= 2 then
                        recoveringPlayers[Player.UserId] = true
                    elseif currentHealth >= 40 then
                        recoveringPlayers[Player.UserId] = nil
                    end
                    
                    -- Only scales back up if above 2 HP AND completely out of the recovering state
                    if currentHealth > 2 and not recoveringPlayers[Player.UserId] then
                        if hitboxEnabled then
                            HRP.Size = Vector3.new(sizeX, sizeY, sizeZ)
                        else
                            HRP.Size = Vector3.new(2, 2, 1)
                        end
                        
                        HRP.Transparency = hitBoxTransparency
                        HRP.CanCollide = false
                        HRP.CanTouch = true
                        HRP.Massless = false
                        HRP.Color = Color3.fromRGB(255, 0, 0)
                        
                        local lineThickness = (1 - hitBoxTransparency) * 0.08
                        local areLinesVisible = hitBoxTransparency < 1
                        
                        local outline = HRP:FindFirstChild('SelectionBox')
                        if not outline then
                            outline = Instance.new('SelectionBox')
                            outline.Name = 'SelectionBox'
                            outline.Parent = HRP
                            outline.Adornee = HRP
                            outline.Color3 = Color3.fromRGB(0, 0, 0)
                        end
                        outline.LineThickness = lineThickness
                        outline.Visible = areLinesVisible
                    else
                        -- Drops down to a stable 1,1,1 structure right on their core axis when knocked
                        HRP.Size = Vector3.new(1, 1, 1)
                        HRP.Transparency = 1
                        HRP.CanCollide = false
                        HRP.CanTouch = false
                        HRP.Massless = true
                        
                        local outline = HRP:FindFirstChild('SelectionBox')
                        if outline then
                            outline:Destroy()
                        end
                    end
                end
            end
        end
    end
end)
